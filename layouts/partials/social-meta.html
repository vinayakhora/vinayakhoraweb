{{- /* Open Graph + Twitter Card meta tags. Replaces Hugo's internal opengraph.html / twitter_cards.html, which ignore the Anatole `thumbnail` front-matter param and emit a single site-wide image. WhatsApp will not render a WebP preview image, so the thumbnail is converted to JPEG at build time. Source images are mounted from static/images into assets/images (see config/_default/hugo.toml) so Hugo can process them. */ -}} {{- $title := cond (eq .Kind "home") site.Title (printf "%s | %s" .Title site.Title) -}} {{- $desc := "" -}} {{- with .Params.description }}{{ $desc = . }}{{ else }}{{ with .Description }}{{ $desc = . }}{{ else }}{{ with .Summary }}{{ $desc = . }}{{ else }}{{ $desc = site.Params.description }}{{ end }}{{ end }}{{ end -}} {{- $desc = $desc | plainify | htmlUnescape | chomp -}} {{- /* Resolve the OG image: post thumbnail -> site fallback */ -}} {{- $src := "" -}} {{- with .Params.thumbnail }}{{ $src = . }}{{ else }}{{ with site.Params.ogImage }}{{ $src = . }}{{ else }}{{ $src = "images/logo.png" }}{{ end }}{{ end -}} {{- $src = strings.TrimPrefix "/" $src -}} {{- $ogURL := "" -}} {{- $ogW := 0 -}} {{- $ogH := 0 -}} {{- with resources.Get $src -}} {{- $jpg := . -}} {{- if ne .MediaType.SubType "jpeg" -}} {{- $jpg = .Fit "1200x630 jpg q85" -}} {{- else if gt .Width 1200 -}} {{- $jpg = .Fit "1200x630 jpg q85" -}} {{- end -}} {{- $ogURL = $jpg.Permalink -}} {{- $ogW = $jpg.Width -}} {{- $ogH = $jpg.Height -}} {{- end -}} {{- with $ogURL }} {{- end }} {{- if eq .Kind "page" }} {{- with .Lastmod }} {{- end }} {{- with .Params.categories }} {{- range . }} {{- end }} {{- end }} {{- with .Params.tags }} {{- range first 6 . }} {{- end }} {{- end }} {{- end }} {{- with $ogURL }} {{- end }}
